{{/* Override of the theme partial. Two changes: 1. Supports an `svg` key for marks that Font Awesome 5.15.1 and academicons do not ship (X, Bluesky -- both postdate FA 5.15.1). 2. Drops the theme's aria-hidden="true" on the wrapper and gives each link a real accessible label from `name`. These are links, not decoration, so hiding them from assistive tech loses the only route to them. A scheme-ful url (mailto:, https:) is passed through untouched; only a scheme-less one is treated as site-relative. */}}
{{ range site.Params.social }} {{ $link := .url }} {{ $scheme := (urls.Parse $link).Scheme }} {{ $target := "" }} {{ if not $scheme }} {{ $link = .url | relLangURL }} {{ else if in (slice "http" "https") $scheme }} {{ $target = "target=\"_blank\" rel=\"me noopener\"" }} {{ end }} {{ $label := or .name .icon .svg }} {{- if .svg -}} {{- if eq .svg "x" -}} {{- else if eq .svg "bluesky" -}} {{- end -}} {{- else -}} {{- $pack := or .icon_pack "fas" -}} {{- $pack_prefix := $pack -}} {{- if in (slice "fab" "fas" "far" "fal") $pack -}}{{- $pack_prefix = "fa" -}}{{- end -}} {{- end -}} {{ end }}
